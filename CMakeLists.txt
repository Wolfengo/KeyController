cmake_minimum_required(VERSION 3.25)
project(ssh-keys-prompt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg)
find_package(LayerShellQt 6.6 REQUIRED)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOMOC ON)
add_executable(ssh-keys-prompt ui/prompt.cpp ui/secret_line_edit.h ui/prompt.qrc)
target_link_libraries(ssh-keys-prompt PRIVATE Qt6::Widgets Qt6::Svg LayerShellQt::Interface)
target_compile_options(ssh-keys-prompt PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
target_link_options(ssh-keys-prompt PRIVATE -Wl,-z,relro,-z,now)
install(TARGETS ssh-keys-prompt DESTINATION lib/ssh-keys RENAME prompt)
include(CTest)
if(BUILD_TESTING)
  enable_language(C)
  find_package(Qt6 REQUIRED COMPONENTS Test DBus WaylandCompositor WaylandCompositorPrivate)
  add_executable(prompt-test tests/prompt.cpp ui/secret_line_edit.h ui/prompt.qrc)
  target_link_libraries(prompt-test PRIVATE Qt6::Widgets Qt6::Svg Qt6::Test LayerShellQt::Interface)
  target_compile_options(prompt-test PRIVATE -UNDEBUG)
  add_executable(prompt-preview tests/prompt-preview.cpp ui/secret_line_edit.h ui/prompt.qrc)
  target_link_libraries(prompt-preview PRIVATE Qt6::Widgets Qt6::Svg LayerShellQt::Interface)
  add_executable(prompt-input-method-test tests/prompt-input-method.cpp ui/secret_line_edit.h ui/prompt.qrc)
  target_link_libraries(prompt-input-method-test PRIVATE Qt6::Widgets Qt6::Svg Qt6::Test Qt6::WaylandCompositorPrivate LayerShellQt::Interface)
  target_compile_options(prompt-input-method-test PRIVATE -UNDEBUG)
  add_test(NAME prompt-input-method COMMAND prompt-input-method-test)
  set_tests_properties(prompt-input-method PROPERTIES TIMEOUT 45)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
  add_executable(prompt-accessibility-test tests/prompt-accessibility.cpp ui/secret_line_edit.h ui/prompt.qrc)
  target_link_libraries(prompt-accessibility-test PRIVATE Qt6::Widgets Qt6::Svg Qt6::Test Qt6::DBus Qt6::WaylandCompositor LayerShellQt::Interface)
  add_executable(prompt-accessibility-listener tests/prompt-accessibility-listener.c)
  target_link_libraries(prompt-accessibility-listener PRIVATE PkgConfig::ATSPI)
  add_test(NAME prompt-accessibility COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/prompt-accessibility.py" $<TARGET_FILE:prompt-accessibility-test> $<TARGET_FILE:prompt-accessibility-listener>)
  set_tests_properties(prompt-accessibility PROPERTIES TIMEOUT 45)
  add_test(NAME prompt-accessibility-preflight COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/prompt-accessibility-preflight.py")
  add_test(NAME prompt-consent COMMAND prompt-test)
  set_tests_properties(prompt-consent PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion")
endif()
